cmake_minimum_required(VERSION 3.16)
project(rmtc_bot)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(CURL REQUIRED)
find_package(Threads REQUIRED)
add_library(rmtc_core src/core.cpp src/stage1_planner.cpp src/stage1_joint.cpp src/stage1_subgoal.cpp)
target_include_directories(rmtc_core PUBLIC include)
target_link_libraries(rmtc_core PUBLIC nlohmann_json::nlohmann_json)
add_library(rmtc_ai src/ai.cpp)
target_include_directories(rmtc_ai PUBLIC include)
target_link_libraries(rmtc_ai PUBLIC rmtc_core CURL::libcurl Threads::Threads)
add_executable(contestant src/node.cpp)
ament_target_dependencies(contestant rclcpp std_msgs)
target_link_libraries(contestant rmtc_core rmtc_ai)
# Explicit opt-in build/run: makes paid real API calls, never part of colcon test.
add_executable(ai_benchmark EXCLUDE_FROM_ALL tools/ai_benchmark.cpp)
target_link_libraries(ai_benchmark rmtc_ai)
add_executable(event_revision_benchmark EXCLUDE_FROM_ALL tools/event_revision_benchmark.cpp)
target_link_libraries(event_revision_benchmark rmtc_ai)
install(TARGETS contestant DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(core_test test/core_test.cpp)
  target_link_libraries(core_test rmtc_core)
  ament_add_gtest(ai_test test/ai_test.cpp)
  target_link_libraries(ai_test rmtc_ai)
endif()
ament_package()

# Fast development checks only; never installed into the contestant entrypoint.
add_executable(stage1_route_benchmark EXCLUDE_FROM_ALL tools/stage1_route_benchmark.cpp)
target_link_libraries(stage1_route_benchmark rmtc_core)
add_executable(stage1_own_replay EXCLUDE_FROM_ALL tools/stage1_own_replay.cpp)
target_link_libraries(stage1_own_replay rmtc_core)
